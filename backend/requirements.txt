import { useState, useEffect } from 'react';
import coverBg from '@/assets/cover-bg.png';

interface CoverProps {
  onOpen: () => void;
}

export default function Cover({ onOpen }: CoverProps) {
  const [guestName, setGuestName] = useState('');
  const [isOpening, setIsOpening] = useState(false);
  const [step, setStep] = useState(0);

  useEffect(() => {
    const params = new URLSearchParams(window.location.search);
    const name = params.get('to');
    if (name) setGuestName(name.replace(/\+/g, ' '));
    // stagger elements
    const t1 = setTimeout(() => setStep(1), 400);
    const t2 = setTimeout(() => setStep(2), 800);
    const t3 = setTimeout(() => setStep(3), 1200);
    const t4 = setTimeout(() => setStep(4), 1600);
    return () => { clearTimeout(t1); clearTimeout(t2); clearTimeout(t3); clearTimeout(t4); };
  }, []);

  const handleOpen = () => {
    setIsOpening(true);
    setTimeout(() => onOpen(), 1000);
  };

  return (
    <div className={`fixed inset-0 z-50 flex items-center justify-center transition-all duration-1000 ${isOpening ? 'opacity-0 scale-110 pointer-events-none' : 'opacity-100'
      }`}>
      {/* Background Photo */}
      <div className="absolute inset-0 bg-parallax" style={{
        backgroundImage: `url(${coverBg})`,
        backgroundSize: 'cover',
        backgroundPosition: 'center',
      }} />
      {/* Overlay */}
      <div className="absolute inset-0" style={{
        background: 'linear-gradient(180deg, rgba(249,246,241,0.82) 0%, rgba(240,235,227,0.75) 40%, rgba(245,230,224,0.85) 100%)',
      }} />

      {/* Animated falling leaves */}
      <div className="absolute inset-0 overflow-hidden pointer-events-none">
        {[...Array(6)].map((_, i) => (
          <div key={i} className="leaf" style={{
            left: `${15 + i * 14}%`,
            animationDuration: `${10 + i * 3}s`,
            animationDelay: `${i * 2}s`,
            top: '-20px',
          }}>
            <svg width="16" height="16" viewBox="0 0 16 16" fill="none" style={{
              opacity: 0.25,
              transform: `rotate(${i * 45}deg)`,
            }}>
              <path d="M8 0C8 0 2 6 2 10C2 13 5 16 8 16C11 16 14 13 14 10C14 6 8 0 8 0Z"
                fill="var(--wedding-gold)" />
            </svg>
          </div>
        ))}
      </div>

      {/* Glowing orbs */}
      <div className="absolute inset-0 pointer-events-none overflow-hidden">
        <div className="absolute top-[15%] left-[20%] w-32 h-32 rounded-full opacity-[0.12]" style={{
          background: 'radial-gradient(circle, var(--wedding-gold) 0%, transparent 70%)',
          animation: 'gentlePulse 5s ease-in-out infinite',
        }} />
        <div className="absolute bottom-[20%] right-[15%] w-40 h-40 rounded-full opacity-[0.10]" style={{
          background: 'radial-gradient(circle, var(--wedding-sage) 0%, transparent 70%)',
          animation: 'gentlePulse 6s ease-in-out infinite 1s',
        }} />
      </div>

      {/* Content */}
      <div className="relative z-10 text-center px-8 max-w-md w-full">
        {/* Animated ornament */}
        <div className={`mb-5 transition-all duration-1000 ${step >= 1 ? 'opacity-100 translate-y-0' : 'opacity-0 -translate-y-6'}`}>
          <svg className="mx-auto" width="100" height="40" viewBox="0 0 100 40" fill="none" style={{
            animation: step >= 1 ? 'sway 6s ease-in-out infinite' : 'none',
          }}>
            <path d="M50 35C38 26 15 24 3 27" stroke="var(--wedding-gold)" strokeWidth="0.8" fill="none" opacity="0.5" />
            <path d="M50 35C62 26 85 24 97 27" stroke="var(--wedding-gold)" strokeWidth="0.8" fill="none" opacity="0.5" />
            <path d="M50 35C40 28 22 28 10 31" stroke="var(--wedding-gold)" strokeWidth="0.8" fill="none" opacity="0.3" />
            <path d="M50 35C60 28 78 28 90 31" stroke="var(--wedding-gold)" strokeWidth="0.8" fill="none" opacity="0.3" />
            <path d="M50 35C44 30 30 31 18 33" stroke="var(--wedding-gold)" strokeWidth="0.6" fill="none" opacity="0.2" />
            <path d="M50 35C56 30 70 31 82 33" stroke="var(--wedding-gold)" strokeWidth="0.6" fill="none" opacity="0.2" />
            <circle cx="50" cy="35" r="2.5" fill="var(--wedding-gold)" opacity="0.4" />
          </svg>
        </div>

        {/* "The Wedding Of" */}
        <p className={`font-ui text-[13px] md:text-[15px] tracking-[0.45em] uppercase mb-5 transition-all duration-900 ${step >= 1 ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-4'
          }`} style={{ color: 'var(--wedding-gold)' }}>
          The Wedding Of
        </p>

        {/* Names with shimmer */}
        <h1 className={`font-script text-[70px] md:text-[76px] lg:text-[80px] leading-[1.05] mb-1 transition-all duration-900 ${step >= 2 ? 'opacity-100 translate-y-0 shimmer-text' : 'opacity-0 translate-y-4'
          }`}>
          Yuven
        </h1>
        <p className={`font-script text-[40px] md:text-[34px] my-1 transition-all duration-900 ${step >= 2 ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-4'
          }`} style={{ color: 'var(--wedding-gold)', animation: step >= 2 ? 'float 4s ease-in-out infinite' : 'none' }}>
          &amp;
        </p>
        <h1 className={`font-script text-[70px] md:text-[76px] lg:text-[80px] leading-[1.05] mb-6 transition-all duration-900 ${step >= 2 ? 'opacity-100 translate-y-0 shimmer-text' : 'opacity-0 translate-y-4'
          }`} style={{ animationDelay: '0.3s' }}>
          Riani
        </h1>

        {/* Animated divider */}
        <div className={`ornament-divider ornament-divider-animated mb-5 transition-all duration-700 ${step >= 2 ? 'opacity-100' : 'opacity-0'
          }`}>
          <span className={step >= 2 ? '' : ''} style={{
            color: 'var(--wedding-gold)', fontSize: '8px',
            animation: step >= 2 ? 'float 3s ease-in-out infinite' : 'none',
          }}>◆</span>
        </div>

        {/* Guest name */}
        {guestName && (
          <div className={`mb-5 transition-all duration-900 ${step >= 3 ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-4'
            }`}>
            <p className="font-ui text-[12px] md:text-[10px] tracking-[0.25em] uppercase mb-1.5" style={{ color: 'var(--wedding-text-light)' }}>
              Kepada Yth.
            </p>
            <p className="font-body text-lg md:text-xl font-semibold" style={{ color: 'var(--wedding-charcoal)' }}>
              {guestName}
            </p>
          </div>
        )}

        {/* Date */}
        <p className={`font-ui text-[11px] md:text-[11px] tracking-[0.3em] mb-7 transition-all duration-900 ${step >= 3 ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-4'
          }`} style={{ color: 'var(--wedding-text-light)' }}>
          24 · 04 · 2026
        </p>

        {/* Button */}
        <button onClick={handleOpen}
          className={`font-ui text-[12px] !mt-[1rem] !px-[4px] !py-[2px] md:text-[12px] tracking-[0.2em] uppercase px-10 py-3.5 rounded-full border-2 cursor-pointer transition-all duration-900 hover:bg-[var(--wedding-gold)] hover:text-white hover:border-[var(--wedding-gold)] ${step >= 4 ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-4'
            }`}
          style={{
            borderColor: 'var(--wedding-gold)',
            color: 'var(--wedding-gold-dark)',
            background: 'rgba(255,255,255,0.3)',
            backdropFilter: 'blur(6px)',
            animation: step >= 4 ? 'pulse 2.5s infinite' : 'none',
            
          }}>
          Buka Undangan
        </button>
      </div>
    </div>
  );
}
